cmake_minimum_required(VERSION 3.20)
project(DK64Coop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(PORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Prepared DK64 Android port workspace")
option(COOP_SANITIZE "Build host tests with address/undefined sanitizers" OFF)
add_library(coop_core STATIC native/protocol.cpp native/session.cpp native/progress.cpp native/animation.cpp native/combat.cpp native/items.cpp native/world.cpp native/transient.cpp)
set_target_properties(coop_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(coop_core PUBLIC native)
if(WIN32)
    target_link_libraries(coop_core PUBLIC ws2_32 iphlpapi)
    if(MINGW)
        # Distribute one DLL, with no extra libc++/unwind DLLs beside the mod.
        target_link_options(coop_core PUBLIC -static-libstdc++ -static-libgcc)
    endif()
endif()
if(MSVC)
    target_compile_options(coop_core PRIVATE /W4)
else()
    target_compile_options(coop_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(COOP_SANITIZE AND NOT WIN32 AND NOT ANDROID)
    target_compile_options(coop_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(coop_core PUBLIC -fsanitize=address,undefined)
endif()
add_library(dk64_coop_bridge SHARED native/bridge.cpp)
set_target_properties(dk64_coop_bridge PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(dk64_coop_bridge PRIVATE "${PORT_ROOT}/upstream/lib/N64ModernRuntime/N64Recomp/include")
target_link_libraries(dk64_coop_bridge PRIVATE coop_core)
if(ANDROID)
    target_link_options(dk64_coop_bridge PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
endif()
add_executable(coop_peer tools/peer.cpp)
target_link_libraries(coop_peer PRIVATE coop_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(coop_items_tests tests/items_tests.cpp)
    target_link_libraries(coop_items_tests PRIVATE coop_core)
    add_test(NAME coop_collectible_flags_and_recovery COMMAND coop_items_tests)
    add_executable(coop_combat_actor_tests tests/combat_actor_tests.cpp)
    target_link_libraries(coop_combat_actor_tests PRIVATE coop_core)
    add_test(NAME coop_combat_game_adapter COMMAND coop_combat_actor_tests)
    add_executable(coop_combat_tests tests/combat_tests.cpp)
    target_link_libraries(coop_combat_tests PRIVATE coop_core)
    add_test(NAME coop_combat_shots_and_defeat_authority COMMAND coop_combat_tests)
    add_executable(coop_animation_tests tests/animation_tests.cpp)
    target_link_libraries(coop_animation_tests PRIVATE coop_core)
    add_test(NAME coop_animation_pose_and_timeline COMMAND coop_animation_tests)
    add_executable(coop_tests tests/core_tests.cpp)
    target_link_libraries(coop_tests PRIVATE coop_core)
    add_test(NAME coop_protocol_and_sessions COMMAND coop_tests)
    add_executable(coop_bridge_tests tests/bridge_tests.cpp)
    target_include_directories(coop_bridge_tests PRIVATE "${PORT_ROOT}/upstream/lib/N64ModernRuntime/N64Recomp/include")
    target_link_libraries(coop_bridge_tests PRIVATE dk64_coop_bridge coop_core)
    add_test(NAME coop_native_abi COMMAND coop_bridge_tests)
    add_executable(coop_progress_tests tests/progress_tests.cpp)
    target_link_libraries(coop_progress_tests PRIVATE coop_core)
    add_test(NAME coop_reserved_progress_wire_regression COMMAND coop_progress_tests)
    add_executable(coop_transition_tests tests/transition_tests.cpp)
    add_test(NAME coop_coordinated_transitions COMMAND coop_transition_tests)
    add_executable(coop_transient_tests tests/transient_tests.cpp)
    target_link_libraries(coop_transient_tests PRIVATE coop_core)
    add_test(NAME coop_same_area_transient_authority COMMAND coop_transient_tests)
    add_executable(coop_transient_game_tests tests/transient_game_tests.cpp)
    add_test(NAME coop_same_area_game_adapter COMMAND coop_transient_game_tests)
    add_executable(coop_trace_tests tests/trace_tests.cpp)
    target_link_libraries(coop_trace_tests PRIVATE coop_core)
    add_test(NAME coop_private_lan_trace_endpoint COMMAND coop_trace_tests)
    add_executable(coop_menu_slot_policy_tests tests/menu_slot_policy_tests.cpp)
    add_test(NAME coop_single_slot_menu_policy COMMAND coop_menu_slot_policy_tests)
    add_executable(coop_lifecycle_policy_tests tests/lifecycle_policy_tests.cpp)
    add_test(NAME coop_upstream_lifecycle_events COMMAND coop_lifecycle_policy_tests)
    add_executable(coop_recovery_policy_tests tests/recovery_policy_tests.cpp)
    add_test(NAME coop_manual_host_recovery COMMAND coop_recovery_policy_tests)
endif()
