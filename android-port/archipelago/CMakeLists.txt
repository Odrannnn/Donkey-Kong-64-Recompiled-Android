cmake_minimum_required(VERSION 3.20)
project(DK64Archipelago LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(PORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Prepared Android port workspace")
option(AP_SANITIZE "Host address/undefined behavior sanitizers" OFF)
set(AP_DEPS "${PORT_ROOT}/.local" CACHE PATH "Pinned native dependencies from tools/bootstrap-native.py")
file(READ "${AP_DEPS}/ap-ixwebsocket/ixwebsocket/IXWebSocketTransport.cpp" IX_TRANSPORT)
if(NOT IX_TRANSPORT MATCHES "DK64 AP: bounded frames")
    message(FATAL_ERROR "Run tools/bootstrap-native.py before configuring")
endif()
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
set(DISABLE_PACKAGE_CONFIG_AND_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${AP_DEPS}/ap-mbedtls" mbedtls EXCLUDE_FROM_ALL)
set(USE_TLS ON CACHE BOOL "" FORCE)
set(USE_MBED_TLS ON CACHE BOOL "" FORCE)
set(USE_OPEN_SSL OFF CACHE BOOL "" FORCE)
set(USE_ZLIB OFF CACHE BOOL "" FORCE)
set(IXWEBSOCKET_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FOUND TRUE)
set(MBEDTLS_INCLUDE_DIRS "${AP_DEPS}/ap-mbedtls/include")
set(MBEDTLS_LIBRARIES mbedtls mbedx509 mbedcrypto)
add_subdirectory("${AP_DEPS}/ap-ixwebsocket" ixwebsocket EXCLUDE_FROM_ALL)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/vendor/cacert.pem" AP_CA_BUNDLE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/vendor/cacert.pem"
    "${CMAKE_CURRENT_SOURCE_DIR}/generated/catalog.json")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ca_bundle.hpp" "#pragma once\nnamespace dkap { inline constexpr char ca_bundle[] = R\"APCA(${AP_CA_BUNDLE})APCA\"; }\n")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/generated/catalog.json" AP_CATALOG_JSON)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/campaign_catalog.hpp" "#pragma once\nnamespace dkap { inline constexpr char campaign_catalog_json[] = R\"APCAT(${AP_CATALOG_JSON})APCAT\"; }\n")
if(NOT MSVC)
    set_source_files_properties(mod/campaign_inventory.c PROPERTIES COMPILE_OPTIONS -Wno-unused-variable)
endif()
add_library(ap_core STATIC native/state.cpp native/journal.cpp native/session.cpp native/campaign.cpp native/seed.cpp native/receipts.cpp native/campaign_journal.cpp native/campaign_channel.cpp native/campaign_protocol.cpp native/campaign_session.cpp native/scouts.cpp native/rdram.cpp mod/campaign_inventory.c)
target_include_directories(ap_core PUBLIC native vendor PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(ap_core PUBLIC ixwebsocket mbedcrypto)
set_target_properties(ap_core PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(MSVC)
    target_compile_options(ap_core PRIVATE /W4 /WX)
else()
    target_compile_options(ap_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        # Clang supports this embedded PEM literal; the diagnostic is only an ISO minimum guarantee.
        target_compile_options(ap_core PRIVATE -Wno-overlength-strings)
    endif()
endif()
if(WIN32)
    target_link_libraries(ap_core PUBLIC ws2_32)
endif()
if(MINGW)
    target_link_options(ap_core PUBLIC -static-libstdc++ -static-libgcc)
endif()
if(AP_SANITIZE AND NOT ANDROID AND NOT WIN32)
    target_compile_options(ap_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ap_core PUBLIC -fsanitize=address,undefined)
endif()
add_library(dk64_ap_bridge SHARED native/bridge.cpp)
set_target_properties(dk64_ap_bridge PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
target_include_directories(dk64_ap_bridge PRIVATE "${PORT_ROOT}/upstream/lib/N64ModernRuntime/N64Recomp/include")
target_link_libraries(dk64_ap_bridge PRIVATE ap_core)
if(ANDROID)
    target_link_options(dk64_ap_bridge PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
endif()
include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
    add_executable(ap_campaign_tests tests/campaign_tests.cpp)
    target_link_libraries(ap_campaign_tests PRIVATE ap_core)
    target_compile_definitions(ap_campaign_tests PRIVATE AP_CATALOG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/generated/catalog.json")
    add_test(NAME ap_full_catalog_inventory_and_checks COMMAND ap_campaign_tests)
    add_executable(ap_campaign_journal_tests tests/campaign_journal_tests.cpp)
    target_link_libraries(ap_campaign_journal_tests PRIVATE ap_core)
    add_test(NAME ap_campaign_durable_journal COMMAND ap_campaign_journal_tests)
    add_executable(ap_campaign_protocol_tests tests/campaign_protocol_tests.cpp)
    target_link_libraries(ap_campaign_protocol_tests PRIVATE ap_core)
    add_test(NAME ap_campaign_protocol COMMAND ap_campaign_protocol_tests)
    add_executable(ap_tls_probe tests/tls_probe.cpp)
    target_link_libraries(ap_tls_probe PRIVATE ixwebsocket)
    add_executable(ap_peer tools/direct-peer.cpp)
    target_link_libraries(ap_peer PRIVATE ap_core)
    add_executable(ap_campaign_peer tools/campaign-peer.cpp)
    target_link_libraries(ap_campaign_peer PRIVATE ap_core)
    add_executable(ap_tests tests/direct_tests.cpp)
    target_link_libraries(ap_tests PRIVATE ap_core)
    add_test(NAME ap_direct_state_and_grant_policy COMMAND ap_tests)
    add_executable(ap_scout_tests tests/scout_tests.cpp)
    target_link_libraries(ap_scout_tests PRIVATE ap_core)
    add_test(NAME ap_scout_metadata_and_shop_labels COMMAND ap_scout_tests)
    add_executable(ap_bridge_tests tests/bridge_tests.cpp)
    target_include_directories(ap_bridge_tests PRIVATE "${PORT_ROOT}/upstream/lib/N64ModernRuntime/N64Recomp/include")
    target_link_libraries(ap_bridge_tests PRIVATE dk64_ap_bridge ap_core)
    add_test(NAME ap_native_abi_bounds COMMAND ap_bridge_tests)
endif()
