cmake_minimum_required(VERSION 3.20)
project(DK64AndroidAudioTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_VIDEO OFF CACHE BOOL "" FORCE)
set(SDL_RENDER OFF CACHE BOOL "" FORCE)
set(SDL_JOYSTICK OFF CACHE BOOL "" FORCE)
set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
foreach(driver ALSA PULSEAUDIO PIPEWIRE JACK OSS SNDIO DISKAUDIO)
    set(SDL_${driver} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${PORT_ROOT}/.local/deps/SDL" sdl)
add_executable(android_audio_tests AudioTests.cpp "${PORT_ROOT}/platform/android_audio.cpp")
target_include_directories(android_audio_tests PRIVATE "${PORT_ROOT}/platform")
target_link_libraries(android_audio_tests PRIVATE SDL2-static)
target_compile_options(android_audio_tests PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(android_audio_tests PRIVATE -fsanitize=address,undefined
    -Wl,--wrap=SDL_QueueAudio -Wl,--wrap=SDL_ClearQueuedAudio -Wl,--wrap=SDL_GetQueuedAudioSize)
enable_testing()
add_test(NAME android_audio_resume_and_frames COMMAND android_audio_tests)
